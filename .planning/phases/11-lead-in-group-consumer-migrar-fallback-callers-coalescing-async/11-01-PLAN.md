# ============================================================================
# PROBE-COALESCE-01 (Fase 11): asyncio.Lock per-chave evita probe concorrente.
# ============================================================================
# 3 jobs paralelos pro mesmo (empresa, grupo, tel, lid) disparam 1 unica
# chamada HTTP Evolution (coalescing via double-checked locking dentro do lock).
#
# Chave: (empresa_id, grupo_jid, telefone_or_empty, lid_lower) — mesmo formato
# da chave de probe_cache (consistencia).
#
# Cleanup: NAO removemos locks do dict. Sao objetos pequenos (~200 bytes cada),
# ~150 entries esteady-state em prod (30 empresas x 5 grupos ativos). Easypanel
# restarta diariamente — dict zera natural. Cap defensivo opcional em fase
# futura se profiling mostrar crescimento (improvavel).
#
# Meta-lock: protege mutacao do dict _probe_locks (criacao sob demanda do lock
# per-chave). Mantido <1us por aquisicao — overhead negligivel sob 1000 lookups/s.

ProbeLockKey = tuple  # (empresa_id, grupo_jid, telefone_or_empty, lid_lower)

_probe_locks: dict = {}
_probe_locks_dict_lock = asyncio.Lock()


def _make_probe_lock_key(empresa_id: str, grupo_jid: str,
                         telefone, lid) -> ProbeLockKey:
    """Chave: (empresa_id, grupo_jid, telefone_or_empty, lid_or_empty_lower).

    Espelha probe_cache._make_key — consistencia entre cache e coalescing.
    """
    return (empresa_id, grupo_jid, telefone or '', (lid or '').lower())


async def _get_probe_lock(key: ProbeLockKey) -> asyncio.Lock:
    """Retorna lock per-key. Cria sob demanda (thread-safe via meta-lock).

    Multiplas chamadas com a MESMA key retornam o MESMO objeto Lock.
    Chamadas com chaves DIFERENTES retornam Locks diferentes (paralelismo).
    """
    async with _probe_locks_dict_lock:
        lock = _probe_locks.get(key)
        if lock is None:
            lock = asyncio.Lock()
            _probe_locks[key] = lock
        return lock


def _reset_probe_locks_for_tests() -> None:
    """SOMENTE para fixture pytest — limpa dict de locks entre testes.

    NAO chamar de codigo de produção.
    """
    _probe_locks.clear()


```

# MEMB-05 (Fase 11): _query_membership_row — DB lookup helper.
# ============================================================================
# Retorna row mais recente NAO removida (saiu_em IS NULL) por (empresa, grupo)
# match em telefone OU lid.
#
# Fase 13 mexe aqui para incluir saiu_em != null (LEAVE consumer).
# Por enquanto: filtro saiu_em IS NULL e DEFENSIVO — grupos com lead removido
# nao contam como "in_group" automatico.

def _query_membership_row(sb, empresa_id: str, grupo_jid: str,
                          telefone, lid) -> dict:
    """Lookup row mais recente em grupo_membership.

    Pre-condicao: pelo menos um de telefone ou lid e nao-nulo.

    Returns: dict {entrou_em, saiu_em, instance_key, atualizado_em} ou None
             se sem row OU erro DB. Caller distingue via try/except se
             quiser semantica distinta (lead_in_group faz isso).
    """
    try:
        q = sb.table('grupo_membership') \
            .select('entrou_em, saiu_em, instance_key, atualizado_em, telefone, lid') \
            .eq('empresa_id', empresa_id) \
            .eq('grupo_jid', grupo_jid) \
            .is_('saiu_em', 'null') \
            .order('atualizado_em', desc=True) \
            .limit(1)

        # Match por telefone OR lid (pattern validado em grupo_fallback.py:241)
        if telefone and lid:
            q = q.or_(f'telefone.eq.{telefone},lid.eq.{lid}')
        elif telefone:
            q = q.eq('telefone', telefone)
        elif lid:
            q = q.eq('lid', lid)
        else:
            # Precondicao violada — caller deveria ter validado
            return None

        result = q.execute()
        if result.data:
            return result.data[0]
        return None
    except Exception as e:
        print(f"[MEMB-LOOKUP] query_error empresa={empresa_id[:8]} "
              f"grupo={grupo_jid[:24]} erro={e}")
        # Re-raise para lead_in_group distinguir "sem row" (None silencioso)
        # vs "erro DB" (exception → source='unknown')
        raise

    from datetime import datetime as _dt
    from app.services import probe_cache
    from app.services.evolution import verificar_lead_no_grupo

    _start = time.monotonic()

    def _emit_log(result: dict) -> dict:
        """Log [MEMB-LOOKUP] + return result (helper interno)."""
        elapsed_ms = int((time.monotonic() - _start) * 1000)
        print(f"[MEMB-LOOKUP] empresa={empresa_id[:8]} grupo={grupo_jid[:24]} "
              f"tel={telefone or '-'} lid={(lid or '-')[:24]} "
              f"source={result.get('source')} "
              f"in_group={result.get('in_group')} "
              f"instance_match={result.get('instance_key_match')} "
              f"latency_ms={elapsed_ms}")
        return result

    # Precondicao: pelo menos um de telefone OR lid
    if not telefone and not lid:
        return _emit_log({
            'in_group': None,
            'source': 'unknown',
            'instance_key_match': None,
            'last_event_at': None,
        })

    # ─────────────────────────────────────────────────────────────────────
    # STEP 1 — CACHE fast-path
    # ─────────────────────────────────────────────────────────────────────
    cached = probe_cache.get(empresa_id, grupo_jid, telefone, lid)
    if cached is not None:
        return _emit_log({
            'in_group': cached,
            'source': 'cache',
            'instance_key_match': True,  # se cacheado, foi validado em probe/membership previo
            'last_event_at': None,  # cache nao armazena timestamp
        })

    # ─────────────────────────────────────────────────────────────────────
    # STEP 2 — MEMBERSHIP DB query
    # ─────────────────────────────────────────────────────────────────────
    try:
        row = _query_membership_row(sb, empresa_id, grupo_jid, telefone, lid)
    except Exception:
        # Erro DB — distingue de "sem row" (None silencioso)
        return _emit_log({
            'in_group': None,
            'source': 'unknown',
            'instance_key_match': None,
            'last_event_at': None,
        })

    if row is not None:
        # ─────────────────────────────────────────────────────────────────
        # STEP 3 — ORPHAN check (caso Fernanda: instance_key mismatch)
        # ─────────────────────────────────────────────────────────────────
        row_instance_key = (row.get('instance_key') or '').strip()
        current_instance_key = ''
        try:
            current_instance_key = (get_instance_key(sb, empresa_id) or '').strip()
        except Exception as _e_ik:
            print(f"[MEMB-LOOKUP] get_instance_key erro empresa={empresa_id[:8]} "
                  f"erro={_e_ik} (treat as match)")

        # Conservador: sem current_instance_key configurada, trata como match
        # (nao bloqueia se config inexistente — mesma logica do helper Fase 10).
        if not current_instance_key:
            instance_match = True
        elif row_instance_key.lower() == current_instance_key.lower():
            instance_match = True
        else:
            instance_match = False

        if instance_match:
            # Row vivo + instance match → autoritativo
            return _emit_log({
                'in_group': True,
                'source': 'membership',
                'instance_key_match': True,
                'last_event_at': row.get('atualizado_em'),
            })
        else:
            # ORPHAN — caso Fernanda. NAO chama probe (economiza HTTP).
            # Caller decide criar novo grupo (sentinel via source='orphan').
            return _emit_log({
                'in_group': False,
                'source': 'orphan',
                'instance_key_match': False,
                'last_event_at': row.get('atualizado_em'),
            })

    # ─────────────────────────────────────────────────────────────────────
    # STEP 4 — Sem row. Se allow_probe_fallback=False, retorna unknown.
    # ─────────────────────────────────────────────────────────────────────
    if not allow_probe_fallback:
        return _emit_log({
            'in_group': None,
            'source': 'unknown',
            'instance_key_match': None,
            'last_event_at': None,
        })

    # ─────────────────────────────────────────────────────────────────────
    # STEP 5 — PROBE Evolution com COALESCING (PROBE-COALESCE-01)
    # ─────────────────────────────────────────────────────────────────────
    if not all([evo_url, evo_key, evo_inst]):
        # Sem creds Evolution — caller esqueceu de passar. Retorna unknown
        # (defensivo — nao crasha).
        print(f"[MEMB-LOOKUP] missing_evo_creds empresa={empresa_id[:8]} "
              f"grupo={grupo_jid[:24]}")
        return _emit_log({
            'in_group': None,
            'source': 'unknown',
            'instance_key_match': None,
            'last_event_at': None,
        })

    key = _make_probe_lock_key(empresa_id, grupo_jid, telefone, lid)
    lock = await _get_probe_lock(key)

    async with lock:
        # DOUBLE-CHECK CACHE (outro consumer pode ter populado durante espera)
        cached_after_lock = probe_cache.get(empresa_id, grupo_jid, telefone, lid)
        if cached_after_lock is not None:
            return _emit_log({
                'in_group': cached_after_lock,
                'source': 'cache',
                'instance_key_match': True,
                'last_event_at': None,
            })

        # Realmente probe (somos o primeiro)
        try:
            probe_result = await verificar_lead_no_grupo(
                evo_url, evo_key, evo_inst, grupo_jid, telefone or '', lead_lid=lid,
            )
            probe_in_group = probe_result.get('in_group')
        except Exception as _e_probe:
            print(f"[MEMB-LOOKUP] probe_exception empresa={empresa_id[:8]} "
                  f"grupo={grupo_jid[:24]} erro={_e_probe}")
            probe_in_group = None

        # Cacheia resultado (probe_cache.set rejeita None silenciosamente,
        # entao errors nao poluem cache)
        if probe_in_group is not None:
            probe_cache.set(empresa_id, grupo_jid, telefone, lid, probe_in_group)

        return _emit_log({
            'in_group': probe_in_group,
            'source': 'probe',
            'instance_key_match': None,  # n/a — sem row para comparar
            'last_event_at': None,
        })


```
