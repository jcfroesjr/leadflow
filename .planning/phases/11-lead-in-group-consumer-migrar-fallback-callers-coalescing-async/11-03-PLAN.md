@pytest.mark.asyncio
async def test_coalesce_three_concurrent_lookups_one_probe(monkeypatch):
    """3 jobs paralelos pra MESMA chave disparam 1 unica chamada HTTP Evolution.

    Verifica double-checked locking: 1º caller probe, 2º e 3º caller fazem
    re-check cache APÓS adquirir lock e retornam cached.
    """
    from app.services.grupo_membership import lead_in_group

    probe_call_count = 0
    probe_call_args = []

    async def fake_probe(base_url, api_key, instancia, grupo_jid, telefone, lead_lid=""):
        nonlocal probe_call_count
        probe_call_count += 1
        probe_call_args.append((grupo_jid, telefone, lead_lid))
        # Simula latência do probe Evolution
        await asyncio.sleep(0.05)
        return {"ok": True, "in_group": True, "erro": ""}

    monkeypatch.setattr(
        "app.services.grupo_membership.verificar_lead_no_grupo",
        fake_probe,
    )
    # Mock _query_membership_row para retornar None (miss → cai pro probe)
    monkeypatch.setattr(
        "app.services.grupo_membership._query_membership_row",
        lambda *a, **k: None,
    )

    sb_mock = MagicMock()

    # 3 chamadas paralelas, MESMA chave
    args = ("emp-uuid", "5511999999999", "120363aaa@g.us")
    results = await asyncio.gather(
        lead_in_group(sb_mock, *args, lid="", evo_url="x", evo_key="y", evo_inst="z"),
        lead_in_group(sb_mock, *args, lid="", evo_url="x", evo_key="y", evo_inst="z"),
        lead_in_group(sb_mock, *args, lid="", evo_url="x", evo_key="y", evo_inst="z"),
    )

    # Asserts críticos
    assert probe_call_count == 1, f"esperado 1 probe, teve {probe_call_count}"
    assert all(r['in_group'] is True for r in results), "todos retornam True"

    # 1º caller é probe (acquired lock primeiro)
    # 2º e 3º caller são cache (re-check dentro do lock encontra valor)
    sources = sorted(r['source'] for r in results)
    assert sources == ['cache', 'cache', 'probe'], \
        f"esperado ['cache', 'cache', 'probe'], teve {sources}"
