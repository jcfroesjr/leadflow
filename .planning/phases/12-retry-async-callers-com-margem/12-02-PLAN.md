    async def recuperar_probe_retries_pendentes() -> None:

        from datetime import datetime, timedelta
        from app.db.client import get_supabase
        try:
            sb = get_supabase()
        except Exception as _e_sb:
            print(f"[PROBE-RETRY-RECOVERY] erro get_supabase: {_e_sb}")

        janela_fim = now + timedelta(hours=6)

            q = (sb.table("agendamentos")
                 .select("id, empresa_id, telefone, grupo_jid, data_hora_utc")

                 .not_.is_("grupo_jid", "null")
                 .limit(200)
                 .execute())
            rows = q.data or []

        recuperados = 0

                )

            except Exception as _e_item:
                print(f"[PROBE-RETRY-RECOVERY] erro item ag={ag.get('id')}: {_e_item}")
        print(f"[PROBE-RETRY-RECOVERY] re-enfileirados={recuperados} de {len(rows)} candidatos")
