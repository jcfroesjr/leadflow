    ```python
    async def _probe_retry_job(
        empresa_id: str,
        grupo_jid: str,
        telefone: str,
        lid: str,
        attempts_left: int,
        max_age_seconds: int,
        enqueued_at: datetime,
        evo_url: str,
        evo_key: str,
        evo_inst: str,
        agendamento_id: str = "",
    ) -> None:
        """Job APScheduler: verifica se lead entrou no grupo apos probe negativo.

        Ordem: (1) max_age guard [Valquiria] → (2) tabela primeiro [Rosania] →
        (3) probe Evolution fallback → (4) re-enqueue proxima tentativa.
        NAO chama lead_in_group (evita loop) — usa _query_membership_row +
        verificar_lead_no_grupo diretamente.
        """
        from app.db.client import get_supabase
        from app.services import probe_cache

        # GUARD 1 — max_age absoluto (caso Valquiria 47h late)
        age_sec = (datetime.utcnow() - enqueued_at).total_seconds()
        if age_sec > max_age_seconds:
            print(f"[PROBE-RETRY] [JOB_EXPIRED] age={age_sec:.0f}s > "
                  f"max_age={max_age_seconds}s empresa={empresa_id[:8]} "
                  f"grupo={grupo_jid[:24]} tel={telefone} — descartado")
            return

        sb = get_supabase()

        # SHORT-CIRCUIT — tabela primeiro (caso Rosania T+138s)
        try:
            row = _query_membership_row(sb, empresa_id, grupo_jid, telefone, lid)
        except Exception as _e_row:
            print(f"[PROBE-RETRY] erro query membership: {_e_row} — prossegue probe")
            row = None

        if row is not None:
            probe_cache.set(empresa_id, grupo_jid, telefone, lid, True)
            print(f"[PROBE-RETRY] [TABLE_HIT] empresa={empresa_id[:8]} "
                  f"grupo={grupo_jid[:24]} tel={telefone} — row found, no probe")
            return

        # PROBE Evolution fallback (tabela ainda vazia)
        if not all([evo_url, evo_key, evo_inst]):
            print(f"[PROBE-RETRY] sem creds Evolution — encerra empresa={empresa_id[:8]}")
            return
        try:
            from app.services.evolution import verificar_lead_no_grupo
            res = await verificar_lead_no_grupo(
                evo_url, evo_key, evo_inst, grupo_jid, telefone or "", lead_lid=lid,
            )
            in_group = res.get("in_group")
        except Exception as _e_probe:
            print(f"[PROBE-RETRY] probe error: {_e_probe} empresa={empresa_id[:8]}")
            in_group = None

        if in_group is True:
            probe_cache.set(empresa_id, grupo_jid, telefone, lid, True)
            print(f"[PROBE-RETRY] [PROBE_HIT] empresa={empresa_id[:8]} "
                  f"grupo={grupo_jid[:24]} tel={telefone}")
            return

        # Ainda negativo (False ou None): re-enqueue proxima tentativa
        if attempts_left > 0:
            schedule_probe_retry(
                empresa_id, grupo_jid, telefone, lid,
                attempts_left=attempts_left,
                max_age_seconds=max_age_seconds,
                evo_url=evo_url, evo_key=evo_key, evo_inst=evo_inst,
                agendamento_id=agendamento_id,
            )
        else:
            print(f"[PROBE-RETRY] [MAX_ATTEMPTS] empresa={empresa_id[:8]} "
                  f"grupo={grupo_jid[:24]} tel={telefone} — esgotado sem confirmacao")
    ```
